```yaml
  wechat_nickname: "" # 微信昵称
  bilibili: ""
  luogu: ""
  codeforces: ""
  other: ""           # 非联系方式的账号/网名（WorkBuddy ID、公众号署名等）
  # *_nickname 子键按平台扩展（如 bilibili_nickname）；
  # ID 子键可按 vault 本地需要扩展（如某校学号键——学号常同时是校邮箱前缀）
identity: ""          # 一句话身份：学校/年级/专业/角色（如 "北京大学 计算机 2026 级")
met: ""               # 怎么/在哪认识的？（群聊、比赛、夏令营、朋友介绍……）
strength: ""          # 关系强度：自由中文，不校验。参考词表——
                      # 基础层：密友 / 熟人 / 点头之交 / 刚建联 / 不熟→预期升温
                      # 关系类型词可作替代或修饰：恩师、导师级、队友、室友、学弟、网友……
                      # 允许括号注释与 → 升温箭头（如 "潜在队友 → 已确定搭档"）
last_contact: ""      # 最近一次有意义互动：YYYY-MM-DD（可选全角括号注释，如 2026-11-19（群里确认））
                      # 未知就留空——不要写 —/everyday 之类占位词，绝不猜测编造日期
```
